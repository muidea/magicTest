package modelpackage_test

import (
	"bytes"
	"context"
	"encoding/json"
	"os"
	"path/filepath"
	"reflect"
	"strings"
	"testing"

	bc "github.com/muidea/magicBase/pkg/common"
	"github.com/muidea/magicBase/pkg/scripteffect"
	"github.com/muidea/magicOrm/models"
	"github.com/muidea/magicOrm/orm"
	"github.com/muidea/magicOrm/provider"
	pc "github.com/muidea/magicRunner/internal/modules/kernel/panel/pkg/common"
	mp "github.com/muidea/magicRunner/internal/modules/kernel/panel/pkg/modelpackage"
	"github.com/muidea/magicRunner/internal/pkg/applicationpagecapability"
	rc "github.com/muidea/magicRunner/pkg/common"
	"github.com/muidea/magicRunner/pkg/definitionjson"
)

// This overlay invokes the production codec; it registers no production route,
// changes no source file, connects to no database and executes blocks only as pure local calculations.
func TestChessRoomDefinitionTool(t *testing.T) {
	root, out := os.Getenv("CHESS_DEFINITION_ROOT"), os.Getenv("CHESS_TOOL_OUTPUT")
	toolingRoot := os.Getenv("CHESS_TOOLING_ROOT")
	if toolingRoot == "" {
		toolingRoot = root
	}
	readFrom := func(base, name string, target any) {
		t.Helper()
		raw, err := os.ReadFile(filepath.Join(base, name))
		if err != nil {
			t.Fatal(err)
		}
		if _, ok := target.(*pc.ApplicationDefinitionView); ok {
			err = definitionjson.DecodeProjection(raw, target) // Same legacy metadata projection as Parse.
		} else {
			err = definitionjson.Decode(raw, target)
		}
		if err != nil {
			t.Fatalf("%s: %v", name, err)
		}
	}
	read := func(name string, target any) {
		t.Helper()
		readFrom(root, name, target)
	}
	var identityFixture struct {
		Identity    bc.EntityActionIdentity `json:"identity"`
		OperationID string                  `json:"operationId"`
	}
	readFrom(toolingRoot, "tooling/testdata/write-identity.json", &identityFixture)
	identityProbe := &bc.EntityValueWriteRequest{APIVersion: bc.EntityValueWriteAPIVersion, OperationID: identityFixture.OperationID, Identity: identityFixture.Identity, Entity: "/chessRoomBooking/merchant", Operation: "create"}
	if err := identityProbe.Validate(); err != nil {
		t.Fatal("simulator identity does not satisfy Base contract", err)
	}
	identityProbe.OperationID += identityProbe.OperationID
	if identityProbe.Validate() == nil {
		t.Fatal("noncanonical operation identity accepted")
	}
	var application pc.ApplicationDefinitionView
	read("definition/application/metadata.json", &application)
	var pages rc.ApplicationPageDefinition
	read("definition/application/pages/definition.json", &pages)
	entities := []*pc.EntityDefinitionView{}
	paths, err := filepath.Glob(filepath.Join(root, "definition/application/entities/*.json"))
	if err != nil {
		t.Fatal(err)
	}
	for _, name := range paths {
		raw, err := os.ReadFile(name)
		if err != nil {
			t.Fatal(err)
		}
		var e pc.EntityDefinitionView
		if err = definitionjson.DecodeProjection(raw, &e); err != nil {
			t.Fatalf("%s: %v", name, err)
		}
		if failure := bc.ValidateEntityConstraints(&e.Object); failure != nil {
			t.Fatal(failure)
		}
		if failure := e.Object.Verify(); failure != nil {
			t.Fatal(failure)
		}
		if failure := bc.ValidateEntityWriteRules(&e.Object, e.WriteRules); failure != nil {
			t.Fatalf("%s writeRules: %v", e.Name, failure)
		}
		entities = append(entities, &e)
	}
	blockFiles := map[string][]byte{}
	blockRoot := filepath.Join(root, "definition/application/blocks")
	err = filepath.Walk(blockRoot, func(name string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if info.IsDir() {
			return nil
		}
		if !info.Mode().IsRegular() {
			t.Fatalf("non-regular block member: %s", name)
		}
		rel, err := filepath.Rel(blockRoot, name)
		if err != nil {
			return err
		}
		blockFiles[filepath.ToSlash(rel)], err = os.ReadFile(name)
		return err
	})
	if err != nil {
		t.Fatal(err)
	}
	blocks, err := pc.ParseApplicationFunctionBlockFiles(blockFiles)
	if err != nil {
		t.Fatal(err)
	}
	artifact := &pc.ApplicationModelArtifact{
		APIVersion:      mp.ArtifactAPIVersion,
		ApplicationUUID: application.UUID,
		ApplicationKey:  application.Name,
		Version:         pages.ModelRevision,
		RuntimeType:     "model-driven",
		Application:     &application,
		Entities:        entities,
		Pages:           &pages,
		FunctionBlocks:  blocks,
	}
	assetsIndexPath := filepath.Join(root, "definition/assets/index.json")
	if indexBytes, assetErr := os.ReadFile(assetsIndexPath); assetErr == nil {
		var assets pc.ApplicationPackageAssets
		if err = json.Unmarshal(indexBytes, &assets); err != nil {
			t.Fatal(err)
		}
		artifact.Assets = &assets
		if artifact.PackageFiles == nil {
			artifact.PackageFiles = map[string][]byte{}
		}
		for _, item := range assets.Files {
			raw, readErr := os.ReadFile(filepath.Join(root, "definition", filepath.FromSlash(item.Path)))
			if readErr != nil {
				t.Fatal(readErr)
			}
			artifact.PackageFiles[item.Path] = raw
		}
	} else if !os.IsNotExist(assetErr) {
		t.Fatal(assetErr)
	} else {
		t.Fatal("missing definition/assets/index.json")
	}

	built, err := mp.Build(artifact)
	if err != nil {
		t.Fatal("official Build:", err)
	}
	second, err := mp.Build(artifact)
	if err != nil || !bytes.Equal(built.Payload, second.Payload) {
		t.Fatal("Build is not deterministic", err)
	}
	parsed, err := mp.Parse(built.Payload)
	if err != nil {
		t.Fatal("official Parse:", err)
	}
	if !reflect.DeepEqual(parsed.Files, built.Files) {
		t.Fatal("ZIP members changed in Parse")
	}
	if parsed.Artifact.Pages.NavigationMode != rc.PageNavigationModeTree {
		t.Fatalf("expected navigationMode tree, got %q", parsed.Artifact.Pages.NavigationMode)
	}
	if len(parsed.Artifact.Pages.Pages) != 19 {
		t.Fatalf("expected 19 pages, got %d", len(parsed.Artifact.Pages.Pages))
	}
	if err := applicationpagecapability.Validate(parsed.Artifact.Pages); err != nil {
		t.Fatalf("applicationpagecapability.Validate: %v", err)
	}
	runtime := applicationpagecapability.Inspect(parsed.Artifact.Pages)
	if !runtime.Ready {
		raw, _ := json.MarshalIndent(runtime, "", "  ")
		t.Fatalf("page runtime capability: %s", raw)
	}
	if err := pc.FreezeApplicationAuthorization(parsed.Artifact.Application, parsed.Artifact.Entities); err != nil {
		t.Fatalf("pc.FreezeApplicationAuthorization: %v", err)
	}
	if len(parsed.Artifact.Application.Authorization.EntityActions) != 19 {
		t.Fatalf("expected 19 entityActions, got %d", len(parsed.Artifact.Application.Authorization.EntityActions))
	}
	if len(parsed.Artifact.Application.Authorization.Templates) != 5 {
		t.Fatalf("expected 5 CAS templates, got %d", len(parsed.Artifact.Application.Authorization.Templates))
	}
	p := provider.NewRemoteProvider("chess-definition-offline", nil)
	tableModels := []models.Model{}
	fieldCount := 0
	for _, e := range parsed.Artifact.Entities {
		model, failure := p.RegisterModel(&e.Object)
		if failure != nil {
			t.Fatal(failure)
		}
		tableModels = append(tableModels, model)
		fieldCount += len(e.Fields)
	}
	schemaPreflight := map[string]string{}
	for _, model := range tableModels {
		for _, index := range models.GetIndexes(model) {
			for _, name := range index.Fields {
				if !models.IsMainTableField(model, model.GetField(name)) {
					t.Fatalf("index %s references non-main-table field %s.%s", index.Name, model.GetName(), name)
				}
			}
		}
		if failure := orm.PreflightSchemaChange(context.Background(), nil, model, p, "chess_definition_offline"); failure != nil {
			t.Fatal(failure)
		}
		schemaPreflight[model.GetName()] = "official public ORM create preflight passed; no SQL exported or executed"
	}
	projected := []*bc.EntityView{}
	byName := map[string]*bc.EntityView{}
	for _, e := range parsed.Artifact.Entities {
		raw, err := json.Marshal(e)
		if err != nil {
			t.Fatal(err)
		}
		var runtimeEntity bc.EntityView
		if err := definitionjson.DecodeProjection(raw, &runtimeEntity); err != nil {
			t.Fatal(err)
		}
		runtimeEntity.BlockInfo, err = pc.ProjectEntityFunctionBlocks(blocks, e)
		if err != nil {
			t.Fatal(err)
		}
		projected = append(projected, &runtimeEntity)
		byName[e.Name] = &runtimeEntity
	}
	if err := scripteffect.ValidateRuntimeModels(projected); err != nil {
		t.Fatal(err)
	}
	source := byName["bookingSample"]
	positiveCases, negativeCases := 0, 0
	for _, block := range source.BlockInfo {
		targetKey := block.StageInfo.Functions[bc.Insert].CreateEffect.TargetEntity
		target := byName[strings.TrimPrefix(targetKey, "/chessRoomBooking/")]
		plan, err := scripteffect.Freeze(source, target, block)
		if err != nil {
			t.Fatal(err)
		}
		for _, status := range []string{"pending_confirmation", "confirmed", "in_use", "completed", "merchant_rejected", "customer_canceled", "merchant_canceled", "pending_expired", "no_show"} {
			for _, precision := range []string{"", ".000", ".1", ".125"} {
				record := map[string]any{"id": "9007199254740993", "sampleKey": "FIXTURE-SAMPLE-offline-" + status, "merchantId": "9007199254740995", "storeId": "9007199254740997", "roomId": "9007199254740999", "orderId": "9007199254741001", "startAt": "2026-10-26T08:00:00" + precision + "Z", "endAt": "2026-10-26T08:45:00" + precision + "Z", "status": status, "holdUntil": nil, "releasedAt": nil, "releaseReason": nil, "version": 1, "createdAt": "2026-10-26T00:00:00Z", "updatedAt": "2026-10-26T00:00:00Z", "updatedBy": "fixture:operator"}
				active := status == "pending_confirmation" || status == "confirmed" || status == "in_use"
				if status == "pending_confirmation" || status == "pending_expired" {
					record["holdUntil"] = "2026-10-26T07:45:00Z"
				}
				if !active {
					record["releasedAt"] = "2026-10-26T08:45:00" + precision + "Z"
					record["releaseReason"] = "fixture:" + status
				}
				calculate := func(values map[string]any, deployment string) []scripteffect.CandidateCreate {
					raw, err := json.Marshal(values)
					if err != nil {
						t.Fatal(err)
					}
					result, err := plan.Calculate(context.Background(), deployment, raw)
					if err != nil || len(result) != 1 {
						t.Fatalf("%s/%s/%s: %v", block.Key, status, precision, err)
					}
					return result
				}
				candidates := calculate(record, "0123456789abcdef0123456789abcdef")
				if !reflect.DeepEqual(candidates, calculate(record, "0123456789abcdef0123456789abcdef")) {
					t.Fatal("effect calculation is not deterministic")
				}
				if candidates[0].Identity == calculate(record, "1123456789abcdef0123456789abcdef")[0].Identity {
					t.Fatal("derived identity escaped deployment isolation")
				}
				item := candidates[0]
				if item.Entity != targetKey || !strings.HasPrefix(item.Identity, "CRB-S-") {
					t.Fatal("wrong target/identity")
				}
				fields := map[string]bc.EntityValueWriteField{}
				for _, f := range item.Fields {
					fields[f.Name] = f
				}
				if fields["orderId"].Reference == nil || fields["orderId"].Reference.ID != "9007199254741001" {
					t.Fatal("reference precision lost")
				}
				if string(fields["sampleKey"].Value) != `"`+item.Identity+`"` {
					t.Fatal("sample identity mapping mismatch")
				}
				if _, exists := fields["version"]; exists {
					t.Fatal("version must be owner allocated")
				}
				if block.Key == "sample-occupancy" {
					state := "released"
					if active {
						state = "active"
					}
					if string(fields["state"].Value) != `"`+state+`"` || fields["roomId"].Reference.ID != "9007199254740999" || string(fields["allocationRevision"].Value) != "1" {
						t.Fatal("occupancy mapping mismatch")
					}
				} else if string(fields["toStatus"].Value) != `"`+status+`"` || string(fields["eventType"].Value) != `"fixture_snapshot"` || string(fields["operationId"].Value) != `"`+record["sampleKey"].(string)+`"` {
					t.Fatal("event mapping mismatch")
				}
				positiveCases++
				if precision != "" {
					continue
				}
				invalid := []map[string]any{
					{"sampleKey": "production-unmarked"}, {"sampleKey": "FIXTURE-SAMPLE-"}, {"status": "unknown"},
					{"endAt": record["startAt"]}, {"startAt": "2026-02-30T08:00:00Z"},
					{"endAt": "2026-10-26T08:45:00.0001Z"}, {"endAt": "2026-10-26T08:45:00+08:00"},
					{"endAt": "2026-10-26T25:45:00Z"}, {"orderId": "9007199254741001.0"},
				}
				if active {
					invalid = append(invalid, map[string]any{"releasedAt": "2026-10-26T08:45:00Z"}, map[string]any{"releaseReason": "fixture:released"})
				} else {
					invalid = append(invalid, map[string]any{"releasedAt": nil}, map[string]any{"releaseReason": " "}, map[string]any{"releasedAt": "bad"})
				}
				if status == "pending_confirmation" || status == "pending_expired" {
					invalid = append(invalid, map[string]any{"holdUntil": nil}, map[string]any{"holdUntil": "2026-10-26T09:00:00Z"})
				} else {
					invalid = append(invalid, map[string]any{"holdUntil": "2026-10-26T07:45:00Z"})
				}
				for _, changes := range invalid {
					bad := map[string]any{}
					for k, v := range record {
						bad[k] = v
					}
					for k, v := range changes {
						bad[k] = v
					}
					raw, _ := json.Marshal(bad)
					if _, err := plan.Calculate(context.Background(), "0123456789abcdef0123456789abcdef", raw); err == nil {
						t.Fatalf("%s/%s accepted invalid sample: %v", block.Key, status, changes)
					}
					negativeCases++
				}
				raw, _ := json.Marshal(record)
				ctx, cancel := context.WithCancel(context.Background())
				cancel()
				if _, err := plan.Calculate(ctx, "0123456789abcdef0123456789abcdef", raw); err == nil {
					t.Fatal("canceled calculation accepted")
				}
				negativeCases++
			}
		}
	}
	payload := built.Payload
	comparison := "not-requested"
	if target := os.Getenv("CHESS_INSPECT_PACKAGE"); target != "" {
		payload, err = os.ReadFile(target)
		if err != nil {
			t.Fatal(err)
		}
		downloaded, err := mp.Parse(payload)
		if err != nil {
			t.Fatal("download Parse:", err)
		}
		if downloaded.ModelDigest != built.ModelDigest || downloaded.PageDigest != built.PageDigest || downloaded.FunctionBlocksDigest != built.FunctionBlocksDigest {
			t.Fatal("downloaded model/page/block content differs from current definitions")
		}
		parsed = downloaded
		comparison = "model-pages-blocks-match; formal revision identity recorded separately"
	}
	if err = os.MkdirAll(filepath.Join(out, "unpacked"), 0755); err != nil {
		t.Fatal(err)
	}
	for name, raw := range parsed.Files {
		target := filepath.Join(out, "unpacked", filepath.FromSlash(name))
		if err = os.MkdirAll(filepath.Dir(target), 0755); err != nil {
			t.Fatal(err)
		}
		if err = os.WriteFile(target, raw, 0644); err != nil {
			t.Fatal(err)
		}
	}
	unpacked, err := mp.ParseUnpacked(os.DirFS(filepath.Join(out, "unpacked")))
	if err != nil {
		t.Fatal("official ParseUnpacked:", err)
	}
	if !reflect.DeepEqual(unpacked.Artifact.Application.Authorization, application.Authorization) {
		t.Fatal("authorization lost during ZIP/unpacked roundtrip")
	}
	if !reflect.DeepEqual(unpacked.Artifact.Assets, artifact.Assets) {
		t.Fatal("assets lost during ZIP/unpacked roundtrip")
	}
	if unpacked.TransportDigest != "" || unpacked.PackageDigest != parsed.PackageDigest {
		t.Fatal("unpacked digest contract changed")
	}
	if err = os.WriteFile(filepath.Join(out, "package.zip"), payload, 0644); err != nil {
		t.Fatal(err)
	}
	ddlJSON, err := json.MarshalIndent(schemaPreflight, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(out, "schema-preflight.json"), append(ddlJSON, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	blockCount := len(blocks.Index.Definitions)
	assetCount := 0
	if parsed.Artifact.Assets != nil {
		assetCount = len(parsed.Artifact.Assets.Files)
	}
	report := map[string]any{
		"schemaVersion":           "chess-room-booking.source-check/v1",
		"status":                  "offline-definition-validated",
		"applicationKey":          application.Name,
		"applicationUuid":         application.UUID,
		"businessVersion":         application.Version,
		"revisionOrigin":          "local-definition-not-published",
		"packageModelRevision":    parsed.Artifact.Version,
		"entityCount":             len(entities),
		"fieldCount":              fieldCount,
		"pageCount":               len(pages.Pages),
		"dataSourceCount":         len(pages.DataSources),
		"navigationMode":          pages.NavigationMode,
		"entityActionCount":       len(parsed.Artifact.Application.Authorization.EntityActions),
		"templateCount":           len(parsed.Artifact.Application.Authorization.Templates),
		"assetCount":              assetCount,
		"functionBlockCount":      blockCount,
		"runtimeEffectValidation": true,
		"effectCalculationCases":  positiveCases,
		"effectNegativeCases":     negativeCases,
		"enabledCapabilities":     46,
		"plannedCapabilities":     46,
		"gatedCreates":            []string{},
		"officialBuild":           true,
		"officialParse":           true,
		"officialParseUnpacked":   true,
		"deterministicBuild":      true,
		"rebuildBytes":            "passed",
		"postgresDDLGenerated":    false,
		"schemaCreatePreflight":   true,
		"databaseExecuted":        false,
		"pageRuntimeInspection":   runtime,
		"modelDigest":             parsed.ModelDigest,
		"artifactDigest":          parsed.ArtifactDigest,
		"pageDigest":              parsed.PageDigest,
		"functionBlocksDigest":    parsed.FunctionBlocksDigest,
		"packageDigest":           parsed.PackageDigest,
		"transportDigest":         parsed.TransportDigest,
		"comparison":              comparison,
		"sourceChecksOnly":        true,
		"published":               false,
		"installed":               false,
		"businessReady":           false,
		"liveValidation":          "not-run",
	}
	if strings.Contains(os.Getenv("CHESS_TOOL_MODE"), "inspect") {
		report["revisionOrigin"] = "downloaded-package-not-a-publish-proof"
	}
	raw, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(out, "source-checks.json"), append(raw, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}
